// Google Apps Script — Careers / Job Application Form (Frontier One Technology)
// 1. Application form ka data leti hai (JSON POST)
// 2. "Job Applications" Sheet me row append karti hai
// 3. Resume ko Google Drive me save karti hai
// 4. info@ + frontieronetechnology@gmail.com dono ko email bhejti hai

const TO_EMAILS = ["info@frontieronetechnology.com", "frontieronetechnology@gmail.com"];
const SHEET_ID = "1QlYI7sxP-7c7m3l8iysAaTwiqwmJXDWqqX2T_SnAOec";
const DRIVE_FOLDER_ID = "1GiajdFif3dyFq5cz4cofotIyjfh_t6Vr";

function formatPhone(phone) {
  if (!phone) return "";
  const cleaned = phone.replace(/\s+/g, "");
  if (cleaned.startsWith("+")) {
    return cleaned.replace(/\+(\d{2})(\d+)/, "+$1 $2");
  }
  return cleaned;
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet();

    const phone = formatPhone(data.phone);

    // Handle resume — save to Drive with clean naming
    let resumeLink = "";
    if (data.resumeBase64 && data.resumeMime) {
      const clean = function (str) {
        return str.replace(/\s+/g, "_").replace(/[^a-zA-Z0-9_-]/g, "");
      };
      const safeName = clean(data.name) + "_" + clean(data.role || "general") + "_" +
        Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
      const ext = data.resumeName && data.resumeName.includes(".")
        ? "." + data.resumeName.split(".").pop()
        : ".pdf";
      const fileName = safeName + ext;

      const blob = Utilities.newBlob(
        Utilities.base64Decode(data.resumeBase64),
        data.resumeMime,
        fileName
      );
      const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
      const file = folder.createFile(blob);
      resumeLink = file.getUrl();
    }

    // Append row to sheet (phone empty — will write separately as text)
    sheet.appendRow([
      new Date(),              // Timestamp
      data.name || "",         // Full Name
      data.email || "",        // Email
      "",                      // Phone (placeholder)
      data.message || "",      // Cover Message
      data.role || "General",  // Role / Job Title
      resumeLink || "Not provided"  // Resume Link
    ]);

    // Write phone as text with apostrophe so + sign stays
    const lastRow = sheet.getLastRow();
    const phoneCell = sheet.getRange(lastRow, 4);
    phoneCell.setNumberFormat("@");
    phoneCell.setValue(phone);

    // Email notification to BOTH accounts
    const subject = "New Job Application from " + (data.name || "Unknown");
    const body =
      "New Job Application Submission\n\n" +
      "Name: " + (data.name || "N/A") + "\n" +
      "Email: " + (data.email || "N/A") + "\n" +
      "Phone: " + (phone || "N/A") + "\n" +
      "Role: " + (data.role || "General") + "\n" +
      "Cover Message: " + (data.message || "N/A") + "\n" +
      "Resume: " + (resumeLink || "Not provided") + "\n\n" +
      "Submitted: " + new Date().toLocaleString();

    TO_EMAILS.forEach(function (email) {
      MailApp.sendEmail(email, subject, body);
    });

    return ContentService
      .createTextOutput(JSON.stringify({ success: true }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
